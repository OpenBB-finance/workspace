# Semantic Tokens

Semantic tokens are CSS custom properties defined in `src/styles/tokens.css`. They reference the primitive palette and automatically switch between light and dark values. Toggle the theme in the toolbar to see both modes.

export const TokenSwatch = ({ token, label }) => (
  <div style={{ display: "flex", alignItems: "center", gap: 12, padding: "6px 0" }}>
    <div
      style={{
        width: 32,
        height: 32,
        borderRadius: 4,
        background: `var(${token})`,
        border: "1px solid rgba(128,128,128,0.2)",
        flexShrink: 0,
      }}
    />
    <div style={{ display: "flex", flexDirection: "column" }}>
      <code style={{ fontSize: 12 }}>{label || token}</code>
      <span style={{ fontSize: 10, fontFamily: "monospace", opacity: 0.5 }}>var({token})</span>
    </div>
  </div>
);

export const TokenGroup = ({ title, tokens }) => (
  <div style={{ marginBottom: 32 }}>
    <h3>{title}</h3>
    <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(280px, 1fr))", gap: "0 24px" }}>
      {tokens.map((t) => (
        <TokenSwatch key={t} token={t} />
      ))}
    </div>
  </div>
);
